#!/bin/bash

#SBATCH --account=GCAM        # the account that time will be charged to
#SBATCH --partition=slurm     # using the short partition to get the small job in quick; slurm is the normal one to use
#SBATCH --nodes=1             # number of nodes needed; each node contains 64 CPUs and 256 GB of RAM
#SBATCH --time=24:00:00
#SBATCH --array=1-32
#SBATCH --output=logs/%x_%A_%a.out   # separate logs per array task
#SBATCH --error=logs/%x_%A_%a.err

# load R
module purge
module load R/4.4.3

# avoid putting out weird characters
export LANG=en_US.UTF-8
export LC_ALL=en_US.UTF-8
# Also set a stable timezone to avoid timedatectl lookups:
export TZ=UTC

# assign job number to GCAM region number
region=$SLURM_ARRAY_TASK_ID

# R script to run
SCRIPT="/qfs/people/onei736/food_demand/uncertainty/scripts/03_calculate_demand.R"

# run script with the slurm array index as the only argument to the script
date
# time Rscript $RSCRIPT $region
time srun "$(dirname "$(which R)")/Rscript" --vanilla "$SCRIPT" "$region"
